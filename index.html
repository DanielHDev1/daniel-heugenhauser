Code example: quest progression and persistence

This method validates level completion, advances the quest when all of its levels are finished, and persists the updated state. 

public void CompleteLevel(int questID, int levelIndex)
{
    var quest = FindQuestByID(questID);
    if (quest != null)
    {
        var level = quest.GetLevel(levelIndex);
        if (level != null && !level.isLevelCompleted && level.bonesCount > 0)
        {
            level.CompleteLevel();

            if (quest.IsQuestFullyCompleted())
            {
                CompleteQuest();
            }

            SaveManager.SaveQuestData(questList);
        }
        else
        {
            Debug.LogWarning($"Level {levelIndex} was already completed " +
                             $"or not found in quest {questID}.");
        }
    }
    else
    {
        Debug.LogError($"Quest with ID {questID} not found.");
    }
}

Code example: zombie navigation state

The chase state follows the navigation path toward the player and returns the zombie to its idle state when the target is too far away or unavailable. 

public override void Physics_Update(double delta)
{
    if (_player != null)
    {
        _moveSpeed = ZombieManager.chaseSpeed;
        Vector2 playerDirection =
            _player.GlobalPosition - _zombie.GlobalPosition;

        if (playerDirection.Length() > 80)
        {
            ChasePlayer();
        }
        else
        {
            _zombie.Velocity = Vector2.Zero;
        }

        if (playerDirection.Length() > 600)
        {
            _timer.Stop();
            EmitSignal("Transitioned", "idle");
        }
    }
    else
    {
        _timer.Stop();
        EmitSignal("Transitioned", "idle");
    }
}

private void ChasePlayer()
{
    Vector2 nextPathPosition = _navAgent.GetNextPathPosition();
    Vector2 newVelocity =
        (nextPathPosition - _zombie.GlobalPosition).Normalized();

    _zombie.Velocity = newVelocity * _moveSpeed;
}

Code example: adaptive minigame difficulty

Difficulty changes the probability of special mole types and shortens how long each mole remains visible. 

private void SetLevel(int level)
{
    bombRate = Mathf.Min(level * 0.025f, 0.25f);
    hardRate = Mathf.Min(level * 0.025f, 1f);

    float durationMin =
        Mathf.Clamp(1 - level * 0.1f, 0.01f, 1f);
    float durationMax =
        Mathf.Clamp(2 - level * 0.1f, 0.01f, 2f);

    duration = Random.Range(durationMin, durationMax);
}

public void Activate(int level)
{
    grabAnimationDuration = gameManager.grabTimer;
    SetLevel(level);
    CreateNext();
    StartCoroutine(ShowHide(startPosition, endPosition));
}
